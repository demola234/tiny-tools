Add the GitHub check
2 minutes

    

      
Every pull request gets lint, a contract diff comment, and a failing check if it breaks the app without a frontend owner's approval.

      
# .github/workflows/contract.yml
name: Contract
on: pull_request
permissions:
  contents: read
  pull-requests: write
jobs:
  contract:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with: { fetch-depth: 0 }
      - uses: mockmechina/action@v0
        with:
          fail-on-breaking: true
